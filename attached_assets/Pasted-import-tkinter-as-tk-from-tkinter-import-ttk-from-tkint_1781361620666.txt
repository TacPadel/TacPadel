    def next_trainer_question(self):
        self.draw_court_base(self.canvas_t, show_zones=True)
        
        # Phase wechseln, wenn Hauptrunde leer, aber Fehler vorhanden sind
        if not self.main_pool and self.wrong_pool:
            self.main_pool = self.wrong_pool.copy()
            random.shuffle(self.main_pool)
            self.wrong_pool = []
            self.is_retry_phase = True
            messagebox.showinfo("Wiederholungsrunde", "Erste Runde beendet! Jetzt wiederholen wir strukturiert die Szenarien, die noch nicht ganz fehlerfrei waren.")

        # Runde komplett beendet
        if not self.main_pool and not self.wrong_pool:
            self.game_over = True
            self.desc_frame.config(bg="#2ecc71")
            self.desc_label.config(bg="#2ecc71", fg="white", font=("Helvetica", 12, "bold"))
            self.desc_label.config(text=f"🏆 EXZELLENT! Du hast alle Szenarien im Level [{self.current_level}] erfolgreich gemeistert!")
            self.score_label.config(text=f"Erfolgreich gelöst: 9 / 9 — RUNDE BEENDET!")
            self.submit_btn.config(state="disabled", bg="#95a5a6")
            return

        self.current_scenario = self.main_pool.pop(0)
        
        # Design je nach Phase
        if self.is_retry_phase:
            status_prefix = "🔄 [Wiederholung] "
            self.desc_frame.config(bg="#e67e22")
            self.desc_label.config(bg="#e67e22", fg="white")
        else:
            status_prefix = "💡 "
            self.desc_frame.config(bg="#34495e")
            self.desc_label.config(bg="#34495e", fg="#ffffff")
            
        if self.current_level == "Profi":
            self.current_side_profi = random.choice(["Vorhand", "Rückhand"])
            zusatz = f"\n\n[PROFI-INFO] Du wirst auf deiner {self.current_side_profi.upper()}-Seite angespielt!"
            self.desc_label.config(text=status_prefix + self.current_scenario["desc"] + zusatz)
        else:
            self.desc_label.config(text=status_prefix + self.current_scenario["desc"])
            
        self.score_label.config(text=f"Erfolgreich gelöst: {self.score} / 9 | Level: {self.current_level}")
        self.draw_trainer_objects()

    def check_answer(self):
        if self.game_over:
            return
            
        shot = self.shot_var.get()
        zone = self.zone_var.get()
        lauf = self.lauf_var.get()
        s = self.current_scenario
        
        is_correct = False
        
        if self.current_level == "Anfänger":
            if shot in s["valid_shots"]:
                is_correct = True
                messagebox.showinfo("✓ Richtig!", f"Korrekt! Du wählst den {shot}.\n\n{s['explanation']}")
            else:
                messagebox.showerror("Analyse unvollständig", f"Der Schlag {shot} ist hier riskant. Keine Sorge, das Szenario kommt in der Wiederholung noch einmal dran!")
                
        elif self.current_level == "Fortgeschrittener":
            if shot in s["valid_shots"] and zone in s["best_zones"]:
                is_correct = True
                messagebox.showinfo("🎉 Perfekt!", f"Schlag ({shot}) und Platzierung ({zone}) passen genau!\n\n{s['explanation']}")
            elif shot in s["valid_shots"] and zone not in s["best_zones"]:
                messagebox.showwarning("⚠️ Ungenaues Ziel", f"Der Schlag {shot} ist super, aber Zone {zone} ist taktisch unvorteilhaft. (Soll-Zonen: {', '.join(s['best_zones'])})")
            else:
                messagebox.showerror("❌ Fehlentscheidung", f"Ein {shot} ist hier nicht optimal.")
                
        elif self.current_level == "Profi":
            lauf_ok = (lauf == s["lauf_profi"])
            shot_ok = (shot in s["valid_shots"])
            zone_ok = (zone in s["best_zones"])
            
            if lauf_ok and shot_ok and zone_ok:
                is_correct = True
                messagebox.showinfo("★★★ PROFI-PRESTIGE ★★★", f"Weltklasse-Analyse auf der {self.current_side_profi}! Komplette taktische Kette fehlerfrei.")
            else:
                messagebox.showerror("✗ Profi-Fehler", "Taktische Kette unvollständig. Die Situation wird am Ende der Rotation wiederholt.")

        if is_correct:
            self.score += 1
        else:
            self.wrong_pool.append(s)

        self.next_trainer_question()

    def setup_taktikboard(self):
        tk.Label(self.tab_board, text="Freies Taktikboard (Verschiebe die Spieler & Ball)", font=("Helvetica", 14, "bold"), bg="#2c3e50", fg="white").pack(pady=5)
        
        self.canvas_b = tk.Canvas(self.tab_board, width=760, height=360, bg="#2980b9", highlightthickness=2, highlightbackground="#bdc3c7")
        self.canvas_b.pack(pady=5)
        
        self.drag_item = None
        self.drag_data = {"x": 0, "y": 0}
        self.interactive_objects = {}
        
        self.draw_court_base(self.canvas_b, show_zones=True)
        self.spawn_board_objects()
        
        self.canvas_b.bind("<ButtonPress-1>", self.on_press)
        self.canvas_b.bind("<ButtonRelease-1>", self.on_release)
        self.canvas_b.bind("<B1-Motion>", self.on_motion)
        
        tk.Button(self.tab_board, text="Aufstellung zurücksetzen", command=self.spawn_board_objects, bg="#e74c3c", fg="white", font=("Helvetica", 10, "bold")).pack(pady=5)

    def spawn_board_objects(self):
        self.draw_court_base(self.canvas_b, show_zones=True)
        self.interactive_objects.clear()
        
        positions = {
            "p1": {"x": 120, "y": 140, "color": "#e74c3c", "lbl": "DU"},
            "p2": {"x": 120, "y": 240, "color": "#c0392b", "lbl": "PTNER"},
            "g1": {"x": 640, "y": 140, "color": "#f1c40f", "lbl": "GEG1"},
            "g2": {"x": 640, "y": 240, "color": "#f39c12", "lbl": "GEG2"},
            "ball": {"x": 380, "y": 180, "color": "#2ecc71", "lbl": "●"}
        }
        
        for name, data in positions.items():
            size = 14 if name != "ball" else 8
            color_text = "white" if data["color"] != "#f1c40f" and data["color"] != "#f39c12" else "black"
            
            c_id = self.canvas_b.create_oval(data["x"]-size, data["y"]-size, data["x"]+size, data["y"]+size, fill=data["color"], outline="white", tags="drag")
            t_id = self.canvas_b.create_text(data["x"], data["y"], text=data["lbl"], fill=color_text, font=("Helvetica", 8, "bold"), tags="drag")
            
            self.interactive_objects[c_id] = t_id
            self.interactive_objects[t_id] = c_id

    def on_press(self, event):
        items = self.canvas_b.find_closest(event.x, event.y)
        if items and "drag" in self.canvas_b.gettags(items[0]):
            self.drag_item = items[0]
            self.drag_data["x"] = event.x
            self.drag_data["y"] = event.y

    def on_release(self, event):
        self.drag_item = None

    def on_motion(self, event):
        if self.drag_item:
            dx = event.x - self.drag_data["x"]
            dy = event.y - self.drag_data["y"]
            self.canvas_b.move(self.drag_item, dx, dy)
            partner = self.interactive_objects.get(self.drag_item)
            if partner:
                self.canvas_b.move(partner, dx, dy)
            self.drag_data["x"] = event.x
            self.drag_data["y"] = event.y
