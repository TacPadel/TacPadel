    def draw_court_base(self, canvas, show_zones=False):
        canvas.delete("all")
        w, h = 760, 360
        canvas.create_rectangle(10, 10, w-10, h-10, outline="white", width=3)
        net_x = w / 2
        canvas.create_line(net_x, 10, net_x, h-10, fill="#2c3e50", width=4, dash=(4, 2))
        offset_x = 10 + (w - 20) * 0.15
        canvas.create_line(offset_x, 10, offset_x, h-10, fill="white", width=2)
        canvas.create_line(w - offset_x, 10, w - offset_x, h-10, fill="white", width=2)
        canvas.create_line(offset_x, h/2, w - offset_x, h/2, fill="white", width=2)
        canvas.create_line(10, 10, 10, h-10, fill="#7f8c8d", width=6)
        canvas.create_line(w-10, 10, w-10, h-10, fill="#7f8c8d", width=6)
        
        if show_zones:
            for zone, coord in ZONE_COORDS_RIGHT.items():
                canvas.create_text(coord[0], coord[1]-10, text=zone, fill="#34495e", font=("Helvetica", 8, "italic"))
