    def draw_court_base(self, canvas, show_zones=False):
        canvas.delete("all")
        w, h = 760, 360
        canvas.create_rectangle(10, 10, w-10, h-10, outline="white", width=3)
        net_x = w / 2
        canvas.create_line(net_x, 10, net_x, h-10, fill="#2c3e50", width=4, dash=(4, 2))
        offset_x = 10 + (w - 20) * 0.15
        canvas.create_line(offset_x, 10, offset_x, h-10, fill="white", width=2)
        canvas.create_line(w - offset_x, 10, w - offset_x, h-10, fill="white", width=2)
        canvas.create_line(offset_x, h/2, w - offset_x, h/2, fill="white", width=2)
        canvas.create_line(10, 10, 10, h-10, fill="#7f8c8d", width=6)
        canvas.create_line(w-10, 10, w-10, h-10, fill="#7f8c8d", width=6)
        
        if show_zones:
            for zone, coord in ZONE_COORDS_LEFT.items():
                canvas.create_text(coord[0], coord[1]-12, text=zone, fill="#bdc3c7", font=("Helvetica", 8, "bold"))
            for zone, coord in ZONE_COORDS_RIGHT.items():
                canvas.create_text(coord[0], coord[1]-12, text=zone, fill="#34495e", font=("Helvetica", 8, "italic"))

    def next_trainer_question(self):
        self.draw_court_base(self.canvas_t, show_zones=True)
        self.current_scenario = random.choice(self.scenarios)
        
        # Challenge-Farbprüfung zurücksetzen/aktualisieren
        if self.current_level == "Anfänger" and len(self.mastered_shots) >= 9:
            self.desc_frame.config(bg="#f1c40f")
            self.desc_label.config(bg="#f1c40f", fg="black", font=("Helvetica", 11, "bold"))
            challenge_text = f"🏆 GOLD-CHALLENGE BESTANDEN! (9/9 Schläge gelernt)\n\n"
        else:
            self.desc_frame.config(bg="#34495e")
            self.desc_label.config(bg="#34495e", fg="#ffffff", font=("Helvetica", 11, "italic"))
            if self.current_level == "Anfänger":
                challenge_text = f"💡 Progress: {len(self.mastered_shots)}/9 Schläge gemeistert\n\n"
            else:
                challenge_text = ""
        
        if self.current_level == "Profi":
            self.current_side_profi = random.choice(["Vorhand", "Rückhand"])
            zusatz = f"\n\n[PROFI-MODUS] Du wirst auf deiner {self.current_side_profi.upper()}-Seite angespielt!"
            self.desc_label.config(text=challenge_text + self.current_scenario["desc"] + zusatz)
        else:
            self.desc_label.config(text=challenge_text + self.current_scenario["desc"])
            
        self.draw_trainer_objects()

    def check_answer(self):
        shot = self.shot_var.get()
        zone = self.zone_var.get()
        lauf = self.lauf_var.get()
        s = self.current_scenario
        
        self.total += 1
        
        if self.current_level == "Anfänger":
            if shot in s["valid_shots"]:
                self.score += 1
                # Füge den korrekt gelösten Schlag zum Challenge-Set hinzu
                correct_shot_name = s["valid_shots"][0]
                self.mastered_shots.add(correct_shot_name)
                
                if len(self.mastered_shots) == 9:
                    messagebox.showinfo("🏆 GOLD LEVEL!", "Wahnsinn! Du hast alle 9 Padel-Schläge mindestens einmal richtig angewendet! Dein Aufgabenfeld erstrahlt jetzt in Gold.")
                else:
                    messagebox.showinfo("✓ Richtig!", f"Korrekt! In dieser Situation spielst du den {shot}.\n\n{s['explanation']}")
            else:
                messagebox.showerror("✗ Falscher Schlag", f"Leider nicht optimal. Der gewählte Schlag {shot} erhöht das Risiko. Besser: {', '.join(s['valid_shots'])}.")
                
        elif self.current_level == "Fortgeschrittener":
            if shot in s["valid_shots"] and zone in s["best_zones"]:
                self.score += 1
                messagebox.showinfo("🎉 Perfekt!", f"Schlag ({shot}) und Platzierung in Zone {zone} stimmen exakt!\n\n{s['explanation']}")
            elif shot in s["valid_shots"] and zone not in s["best_zones"]:
                messagebox.showwarning("⚠️ Ungenaues Ziel", f"Der Schlag {shot} ist richtig! Die Zone {zone} ist jedoch taktisch unvorteilhaft. Empfohlene Zonen: {', '.join(s['best_zones'])}.")
            else:
                messagebox.showerror("❌ Fehler", f"Fehlentscheidung. Ein {shot} verliert hier den Ball. Richtig wäre: {', '.join(s['valid_shots'])} in die Zonen {', '.join(s['best_zones'])}.")
                
        elif self.current_level == "Profi":
            lauf_ok = (lauf == s["lauf_profi"])
            shot_ok = (shot in s["valid_shots"])
            zone_ok = (zone in s["best_zones"])
            
            fehlerliste = []
            if not lauf_ok: fehlerliste.append(f"- Falscher Stellungs-Laufweg (Soll: {s['lauf_profi']})")
            if not shot_ok: fehlerliste.append(f"- Falscher Schlag auf der {self.current_side_profi} (Empfehlung: {', '.join(s['valid_shots'])})")
            if not zone_ok: fehlerliste.append(f"- Ungünstiges Ziel-Areal (Soll: {', '.join(s['best_zones'])})")
            
            if lauf_ok and shot_ok and zone_ok:
                self.score += 1
                messagebox.showinfo("★★★ PROFI-PRESTIGE ★★★", f"Weltklasse-Analyse auf der {self.current_side_profi}! Komplette taktische Kette fehlerfrei gelöst.")
            else:
                error_msg = "\n".join(fehlerliste)
                messagebox.showerror("✗ Profi-Fehlentscheidung", f"In der Profi-Analyse gab es folgende Mängel:\n\n{error_msg}")
