class FullPadelApp:

        self.root = root
        self.root.title("Padel-Tennis Taktik- & Ausbildungszentrum")
        self.root.geometry("1000x850")
        self.root.configure(bg="#2c3e50")
        
        self.notebook = ttk.Notebook(root)
        self.notebook.pack(expand=True, fill="both", padx=10, pady=10)
        
        style = ttk.Style()
        style.configure("TNotebook", background="#2c3e50")
        style.configure("TNotebook.Tab", font=("Helvetica", 11, "bold"), padding=[15, 5])
        
        self.tab_trainer = tk.Frame(self.notebook, bg="#2c3e50")
        self.tab_board = tk.Frame(self.notebook, bg="#2c3e50")
        self.tab_basics = tk.Frame(self.notebook, bg="#2c3e50")
        
        self.notebook.add(self.tab_trainer, text=" 🎯 Taktiktrainer ")
        self.notebook.add(self.tab_board, text=" 📋 Freies Taktikboard ")
        self.notebook.add(self.tab_basics, text=" 📖 Padel Basics ")
        
        self.setup_taktiktrainer()
        self.setup_taktikboard()
        self.setup_padel_basics()

    def set_level(self, level):
        self.current_level = level
        self.btn_anfaenger.config(bg="#2980b9" if level == "Anfänger" else "#34495e")
        self.btn_fortgeschritten.config(bg="#2980b9" if level == "Fortgeschrittener" else "#34495e")
        self.btn_profi.config(bg="#2980b9" if level == "Profi" else "#34495e")
        
        self.score_label.config(text=f"Score: {self.score} / {self.total} | Level: {self.current_level}")
        self.update_input_fields_visibility()
        self.next_trainer_question()

    def update_input_fields_visibility(self):
        for widget in self.input_frame.winfo_children():
            widget.grid_forget()
            
        if self.current_level == "Anfänger":
            self.lbl_shot.grid(row=0, column=0, padx=5)
            self.shot_menu.grid(row=0, column=1, padx=5)
        elif self.current_level == "Fortgeschrittener":
            self.lbl_shot.grid(row=0, column=0, padx=5)
            self.shot_menu.grid(row=0, column=1, padx=5)
            self.lbl_zone.grid(row=0, column=2, padx=5)
            self.zone_menu.grid(row=0, column=3, padx=5)
        elif self.current_level == "Profi":
            self.lbl_lauf.grid(row=0, column=0, padx=5)
            self.lauf_menu.grid(row=0, column=1, padx=5)
            self.lbl_shot.grid(row=0, column=2, padx=5)
            self.shot_menu.grid(row=0, column=3, padx=5)
            self.lbl_zone.grid(row=0, column=4, padx=5)
            self.zone_menu.grid(row=0, column=5, padx=5)
            
        self.submit_btn.grid(row=0, column=6, padx=15)

    def draw_court_base(self, canvas, show_zones=False):
        canvas.delete("all")
        w, h = 760, 360
        canvas.create_rectangle(10, 10, w-10, h-10, outline="white", width=3)
        net_x = w / 2
        canvas.create_line(net_x, 10, net_x, h-10, fill="#2c3e50", width=4, dash=(4, 2))
        offset_x = 10 + (w - 20) * 0.15
        canvas.create_line(offset_x, 10, offset_x, h-10, fill="white", width=2)
        canvas.create_line(w - offset_x, 10, w - offset_x, h-10, fill="white", width=2)
        canvas.create_line(offset_x, h/2, w - offset_x, h/2, fill="white", width=2)
        canvas.create_line(10, 10, 10, h-10, fill="#7f8c8d", width=6)
        canvas.create_line(w-10, 10, w-10, h-10, fill="#7f8c8d", width=6)
        
        if show_zones:
            for zone, coord in ZONE_COORDS_LEFT.items():
                canvas.create_text(coord[0], coord[1]-12, text=zone, fill="#bdc3c7", font=("Helvetica", 8, "bold"))
            for zone, coord in ZONE_COORDS_RIGHT.items():
                canvas.create_text(coord[0], coord[1]-12, text=zone, fill="#34495e", font=("Helvetica", 8, "italic"))
